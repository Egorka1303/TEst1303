    <script>
        // Инициализация Three.js сцены
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
        
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        container.appendChild(renderer.domElement);

        // Создаем футуристический геометрический 3D-объект (торальный узел / кристалл)
        const geometry = new THREE.IcosahedronGeometry(2, 0);
        const material = new THREE.MeshStandardMaterial({
            color: 0x00f0ff,
            wireframe: true,
            roughness: 0.2,
            metalness: 0.8
        });
        const icosahedron = new THREE.Mesh(geometry, material);
        scene.add(icosahedron);

        // Добавляем плавающие частицы (цифровой шум/техно-пыль)
        const particlesGeometry = new THREE.BufferGeometry();
        const particlesCount = 700;
        const posArray = new Float32Array(particlesCount * 3);

        for(let i = 0; i < particlesCount * 3; i++) {
            posArray[i] = (Math.random() - 0.5) * 15;
        }

        particlesGeometry.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
        const particlesMaterial = new THREE.PointsMaterial({
            size: 0.025,
            color: 0x7928ca
        });
        const particlesMesh = new THREE.Points(particlesGeometry, particlesMaterial);
        scene.add(particlesMesh);

        // Освещение
        const pointLight = new THREE.PointLight(0x00f0ff, 3, 50);
        pointLight.position.set(5, 5, 5);
        scene.add(pointLight);

        const ambientLight = new THREE.AmbientLight(0xffffff, 0.5);
        scene.add(ambientLight);

        camera.position.z = 6;

        // Интерактив: вращение от движения мыши
        let mouseX = 0;
        let mouseY = 0;

        document.addEventListener('mousemove', (event) => {
            mouseX = (event.clientX / window.innerWidth) - 0.5;
            mouseY = (event.clientY / window.innerHeight) - 0.5;
        });

        // Анимационный цикл
        const clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);
            const elapsedTime = clock.getElapsedTime();

            // Плавное вращение 3D-объекта
            icosahedron.rotation.x = elapsedTime * 0.2 + mouseY * 0.5;
            icosahedron.rotation.y = elapsedTime * 0.3 + mouseX * 0.5;

            // Вращение частиц
            particlesMesh.rotation.y = elapsedTime * 0.05;

            renderer.render(scene, camera);
        }

        animate();

        // Адаптивность при изменении размера окна
        window.addEventListener('resize', () => {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        });
    </script>
